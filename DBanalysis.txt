/* Database schema and analysis */

CREATE TABLE Accounts (
    account_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    name TEXT NOT NULL,
    phone_number TEXT,
    account_type TEXT CHECK(account_type IN ('Savings', 'Current')),
    balance DOUBLE NOT NULL DEFAULT 0 CHECK(balance >= 0),
    risk_score INT DEFAULT 0 CHECK(risk_score BETWEEN 0 AND 100),
    open_date TEXT NOT NULL,
    active_status INT DEFAULT 1 CHECK(active_status IN (0,1))
);

CREATE TABLE Transactions (
    transaction_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    sender_account INT,
    receiver_account INT,
    amount DOUBLE NOT NULL CHECK(amount > 0),
    transaction_type TEXT CHECK(transaction_type IN ('deposit','withdrawal','transfer')),
    txn_datetime TEXT NOT NULL,
    flagged INT DEFAULT 0 CHECK(flagged IN (0,1)),
    FOREIGN KEY(sender_account) REFERENCES Accounts(account_id) ON DELETE RESTRICT,
    FOREIGN KEY(receiver_account) REFERENCES Accounts(account_id) ON DELETE RESTRICT,
    CHECK(sender_account IS NOT NULL OR receiver_account IS NOT NULL)
);

CREATE TABLE Devices (
    device_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    account_id INT NOT NULL,
    fingerprint TEXT NOT NULL,
    FOREIGN KEY(account_id) REFERENCES Accounts(account_id) ON DELETE CASCADE
);

CREATE TABLE FraudAlerts (
    alert_id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    transaction_id INT NOT NULL,
    reason TEXT NOT NULL,
    alert_date TEXT NOT NULL,
    FOREIGN KEY(transaction_id) REFERENCES Transactions(transaction_id) ON DELETE CASCADE
);
